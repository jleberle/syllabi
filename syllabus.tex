% Pandoc marks uncaptioned tables with \def\LTcaptype{none}; longtable then

% syllabi no longer need their own \newcounter{none} workaround.

	urlcolor=osuorange,
	plainpages=false,
	]{hyperref}

\titlespacing*{\subsubsection}{0pt}{8pt}{2pt}
